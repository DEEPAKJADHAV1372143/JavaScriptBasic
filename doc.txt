// Sample object
let person = { name: "Deepak", age: 30, city: "Pune" };

// 1. Object.keys, Object.values, Object.entries
console.log(Object.keys(person));   // ["name","age","city"]
console.log(Object.values(person)); // ["Deepak",30,"Pune"]
console.log(Object.entries(person)); // [["name","Deepak"],["age",30],["city","Pune"]]

// 2. Object.assign
let target = { country: "India" };
Object.assign(target, person);
console.log(target); // {country:"India", name:"Deepak", age:30, city:"Pune"}

// 3. Object.create
let proto = { greet() { return "Hello!"; } };
let obj = Object.create(proto);
console.log(obj.greet()); // "Hello!"

// 4. Object.defineProperty, Object.defineProperties
Object.defineProperty(person, "id", { value: 101, writable: false });
console.log(person.id); // 101
Object.defineProperties(person, {
  email: { value: "deepak@example.com", writable: true },
  active: { value: true }
});
console.log(person.email, person.active);

  let count = 0;
  return function() {
    count++;
    return count;
  };

// 2. Prototypes
function Person(name) {
  this.name = name;
}
Person.prototype.sayHello = function() {
  return `Hello, I am ${this.name}`;
};
let p = new Person("Deepak");
console.log(p.sayHello());

// 3. Classes & Inheritance
class Animal {
  constructor(type) { this.type = type; }
  speak() { console.log(`${this.type} makes a sound`); }
}
class Dog extends Animal {
  speak() { console.log("Dog barks"); }
}
new Dog("Dog").speak();

// 8. Proxy
let target = { message: "Hello" };
let proxy = new Proxy(target, {
  get(obj, prop) {
    return prop in obj ? obj[prop] : "Not Found";
  }
});
console.log(proxy.message); // Hello
console.log(proxy.other);   // Not Found
